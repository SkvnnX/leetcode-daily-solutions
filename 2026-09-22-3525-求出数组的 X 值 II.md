## 题意概括

给定正整数数组 `nums` 和正整数 `k`。每个查询 `[index, value, start, x]` 先把 `nums[index]` 永久更新为 `value`，然后临时删除前缀 `nums[0..start-1]`，在剩余数组 `nums[start..n-1]` 中选择删除一个后缀，但删除后必须至少剩下一个元素。

也就是说，对每个查询，需要统计有多少个结束位置 `j` 满足：

```text
product(nums[start..j]) % k == x
```

返回每个查询的统计结果。

## 关键思路

因为 `k <= 5` 很小，可以在线段树的每个节点维护这段区间的两类信息：

1. `prod`：整段区间所有元素乘积对 `k` 取模。
2. `cnt[r]`：这段区间所有非空前缀的乘积模 `k` 等于 `r` 的数量。

设左儿子为 `A`，右儿子为 `B`。合并时：

- 整段乘积：`prod = A.prod * B.prod % k`。
- 新区间的前缀分两种：
  - 完全落在左儿子中：直接继承 `A.cnt`。
  - 跨过左儿子进入右儿子：等于左儿子整段乘积乘上右儿子的某个前缀乘积，所以若右儿子前缀余数为 `r`，合并后余数为 `A.prod * r % k`。

每次查询先做一次单点更新，再在线段树上查询区间 `[start, n)`，答案就是该区间节点信息里的 `cnt[x]`。

## 为什么这样做是正确的

删除前缀后，剩余数组为 `nums[start..n-1]`；再删除一个后缀并保持非空，本质上就是从 `start` 开始选择一个结束位置 `j`，保留连续子数组 `nums[start..j]`。因此查询要数的是区间 `[start, n)` 的所有非空前缀乘积余数分布。

线段树节点维护的 `cnt` 正好描述一段区间内所有非空前缀的乘积余数。合并两个相邻区间时，一个前缀要么在左段内结束，要么包含整个左段并在右段某个前缀结束；上述转移完整且不重不漏。因此区间查询得到的 `cnt[x]` 就是题目要求的 x 值。

## 复杂度分析

- 建树：`O(n * k)`。
- 单点更新：`O(k log n)`。
- 区间查询：`O(k log n)`。
- 总复杂度：`O((n + q log n) * k)`，由于 `k <= 5`，可以视为 `O(n + q log n)`。
- 空间复杂度：`O(n * k)`。

## 容易踩坑点

1. 查询中的更新会永久影响后续查询，不能把每次查询都当作独立数组。
2. 删除后缀后必须剩余非空，所以统计的是 `[start, n)` 的非空前缀，不能包含空数组。
3. 合并线段树节点时要保持左右顺序，乘法虽然数值上交换律成立，但前缀结构不对称。
4. `k` 可以等于 1，此时所有余数都是 0，单位节点的 `prod` 应为 `1 % k`。
5. Python 区间查询需要用左累积和右累积分别维护顺序，最后再合并。

```cpp
class Solution {
    struct Node {
        int prod;
        array<long long, 5> cnt;
    };

    int n, k, size;
    vector<Node> tree;

    Node identity() const {
        Node res;
        res.prod = 1 % k;
        res.cnt.fill(0);
        return res;
    }

    Node makeLeaf(int val) const {
        Node res;
        res.prod = val % k;
        res.cnt.fill(0);
        res.cnt[res.prod] = 1;
        return res;
    }

    Node mergeNode(const Node& a, const Node& b) const {
        Node res;
        res.prod = int(1LL * a.prod * b.prod % k);
        res.cnt = a.cnt;
        for (int r = 0; r < k; ++r) {
            if (b.cnt[r]) {
                int nr = int(1LL * a.prod * r % k);
                res.cnt[nr] += b.cnt[r];
            }
        }
        return res;
    }

    void update(int pos, int val) {
        int p = size + pos;
        tree[p] = makeLeaf(val);
        for (p >>= 1; p; p >>= 1) {
            tree[p] = mergeNode(tree[p * 2], tree[p * 2 + 1]);
        }
    }

    Node query(int l, int r) const {
        Node left = identity(), right = identity();
        l += size;
        r += size;
        while (l < r) {
            if (l & 1) left = mergeNode(left, tree[l++]);
            if (r & 1) right = mergeNode(tree[--r], right);
            l >>= 1;
            r >>= 1;
        }
        return mergeNode(left, right);
    }

public:
    vector<int> resultArray(vector<int>& nums, int K, vector<vector<int>>& queries) {
        k = K;
        n = (int)nums.size();
        size = 1;
        while (size < n) size <<= 1;

        tree.assign(size * 2, identity());
        for (int i = 0; i < n; ++i) tree[size + i] = makeLeaf(nums[i]);
        for (int i = size - 1; i >= 1; --i) {
            tree[i] = mergeNode(tree[i * 2], tree[i * 2 + 1]);
        }

        vector<int> ans;
        ans.reserve(queries.size());
        for (auto& q : queries) {
            int index = q[0], value = q[1], start = q[2], x = q[3];
            nums[index] = value;
            update(index, value);
            ans.push_back((int)query(start, n).cnt[x]);
        }
        return ans;
    }
};
```

```python
from typing import List


class Solution:
    def resultArray(self, nums: List[int], k: int, queries: List[List[int]]) -> List[int]:
        n = len(nums)
        size = 1
        while size < n:
            size <<= 1

        def identity():
            return (1 % k, [0] * k)

        def merge(a, b):
            prod_a, cnt_a = a
            prod_b, cnt_b = b
            cnt = cnt_a[:]
            for r, c in enumerate(cnt_b):
                if c:
                    cnt[(prod_a * r) % k] += c
            return ((prod_a * prod_b) % k, cnt)

        tree = [identity() for _ in range(size * 2)]
        for i, v in enumerate(nums):
            rem = v % k
            cnt = [0] * k
            cnt[rem] = 1
            tree[size + i] = (rem, cnt)

        for i in range(size - 1, 0, -1):
            tree[i] = merge(tree[i * 2], tree[i * 2 + 1])

        def update(pos: int, val: int) -> None:
            rem = val % k
            cnt = [0] * k
            cnt[rem] = 1
            p = size + pos
            tree[p] = (rem, cnt)
            p >>= 1
            while p:
                tree[p] = merge(tree[p * 2], tree[p * 2 + 1])
                p >>= 1

        def query(l: int, r: int):
            left = identity()
            right = identity()
            l += size
            r += size
            while l < r:
                if l & 1:
                    left = merge(left, tree[l])
                    l += 1
                if r & 1:
                    r -= 1
                    right = merge(tree[r], right)
                l >>= 1
                r >>= 1
            return merge(left, right)

        ans = []
        for index, value, start, x in queries:
            nums[index] = value
            update(index, value)
            ans.append(query(start, n)[1][x])
        return ans
```
